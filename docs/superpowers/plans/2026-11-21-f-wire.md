```ts
export type CompleteFn = <T>(settings: LlmSettings, apiKey: string, request: LlmRequest, parse: LlmParse<T>) => Promise<LlmOutcome<T>>;
export const parseNarrationJson: LlmParse<unknown> = raw => { /* JSON.parse, lines 배열 */ };
export function narrationLlmFrom(complete: CompleteFn, current: () => { settings: LlmSettings; apiKey: string }): NarrationLlm {
  return async ({ systemPrompt, userPrompt, schema, timeoutMs, maxOutputTokens }) => {
    const { settings, apiKey } = current();
    const outcome = await complete(settings, apiKey, { overrideSystemPrompt: systemPrompt, overrideUserPrompt: userPrompt, overrideSchema: schema, timeoutMs, ...(maxOutputTokens !== undefined ? { maxOutputTokens } : {}) }, parseNarrationJson);
    return outcome.ok ? { ok: true, raw: outcome.value } : { ok: false, reason: `${outcome.kind}: ${outcome.reason}` };
  };
}
```

```ts
export type DashboardRouteDeps = { dashboard: DashboardData; narrator: Pick<Narrator, "latest" | "replay"> };
export function handleDashboardRoute(method: string | undefined, path: string, request: IncomingMessage, response: ServerResponse, deps: DashboardRouteDeps): boolean {
  if (method === "GET" && path === "/api/dashboard") { deps.dashboard.maybeAutoRefresh(); send(response, 200, { ...deps.dashboard.state(), narration: deps.narrator.latest() }); return true; }
  if (method === "POST" && path === "/api/er/refresh") { send(response, 200, { started: deps.dashboard.refresh() }); return true; }
  if (method === "GET" && path === "/api/dashboard/settings") { send(response, 200, deps.dashboard.settings()); return true; }
  if (method === "PUT" && path === "/api/dashboard/settings") { readBody(request, response, body => void saveSettings(body, response, deps)); return true; }
  if (method === "POST" && path === "/api/narration/replay") { void replay(response, deps); return true; }
  return false;
}
```
